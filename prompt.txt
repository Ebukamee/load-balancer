// Package config loads and validates the load balancer's JSON configuration.
package config

import (
	"encoding/json"
	"errors"
	"fmt"
	"net/url"
	"os"
	"time"
)

// Config is the top-level configuration for the load balancer.
type Config struct {
	ListenAddr  string      `json:"listen_addr"`
	Strategy    string      `json:"strategy"`
	HealthCheck HealthCheck `json:"health_check"`
	Backends    []string    `json:"backends"`
}

// HealthCheck controls how backends are probed (used from Milestone 3).
type HealthCheck struct {
	Path     string   `json:"path"`
	Interval Duration `json:"interval"`
	Timeout  Duration `json:"timeout"`
}

// Duration wraps time.Duration so it can be written as "5s" in JSON.
type Duration struct{ time.Duration }

// UnmarshalJSON parses strings like "500ms" or "5s".
func (d *Duration) UnmarshalJSON(b []byte) error {
	var s string
	if err := json.Unmarshal(b, &s); err != nil {
		return fmt.Errorf("duration must be a string like \"5s\": %w", err)
	}
	v, err := time.ParseDuration(s)
	if err != nil {
		return err
	}
	d.Duration = v
	return nil
}

// Load reads, decodes and validates the config file at path.
func Load(path string) (*Config, error) {
	f, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer f.Close()

	var cfg Config
	dec := json.NewDecoder(f)
	dec.DisallowUnknownFields()
	if err := dec.Decode(&cfg); err != nil {
		return nil, fmt.Errorf("decode %s: %w", path, err)
	}
	if err := cfg.Validate(); err != nil {
		return nil, fmt.Errorf("invalid config: %w", err)
	}
	return &cfg, nil
}

// Validate checks that the config is usable.
func (c *Config) Validate() error {
	if c.ListenAddr == "" {
		return errors.New("listen_addr is required")
	}
	if len(c.Backends) == 0 {
		return errors.New("at least one backend is required")
	}
	for _, raw := range c.Backends {
		u, err := url.Parse(raw)
		if err != nil {
			return fmt.Errorf("backend %q: %w", raw, err)
		}
		if u.Scheme != "http" && u.Scheme != "https" {
			return fmt.Errorf("backend %q: scheme must be http or https", raw)
		}
		if u.Host == "" {
			return fmt.Errorf("backend %q: missing host", raw)
		}
	}
	return nil
}

// Command lb is the load balancer.
package main

import (
	"context"
	"errors"
	"flag"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"loadbalancer/internal/config"
)

func main() {
	configPath := flag.String("config", "configs/config.json", "path to config file")
	flag.Parse()

	logger := slog.New(slog.NewTextHandler(os.Stdout, nil))

	if err := run(*configPath, logger); err != nil {
		logger.Error("fatal", "err", err)
		os.Exit(1)
	}
}

func run(configPath string, logger *slog.Logger) error {
	cfg, err := config.Load(configPath)
	if err != nil {
		return err
	}

	// TODO(M1): build backends from cfg.Backends and forward requests to them
	// instead of returning 503.
	handler := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		http.Error(w, "no backends wired up yet", http.StatusServiceUnavailable)
	})

	srv := &http.Server{
		Addr:              cfg.ListenAddr,
		Handler:           handler,
		ReadHeaderTimeout: 5 * time.Second,
	}

	// Stop cleanly on Ctrl+C / SIGTERM so in-flight requests can finish.
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	errCh := make(chan error, 1)
	go func() {
		logger.Info("load balancer listening", "addr", cfg.ListenAddr, "backends", len(cfg.Backends))
		errCh <- srv.ListenAndServe()
	}()

	select {
	case err := <-errCh:
		if !errors.Is(err, http.ErrServerClosed) {
			return err
		}
	case <-ctx.Done():
		logger.Info("shutting down")
	}

	shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	return srv.Shutdown(shutdownCtx)
}

// Command backend is a tiny HTTP server used as an upstream while developing
// the load balancer. Run several on different ports to see traffic spread out.
package main

import (
	"flag"
	"fmt"
	"log/slog"
	"net/http"
	"os"
	"time"
)

func main() {
	addr := flag.String("addr", ":9001", "listen address")
	name := flag.String("name", "", "name reported in responses (defaults to addr)")
	flag.Parse()
	if *name == "" {
		*name = *addr
	}

	mux := http.NewServeMux()
	mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
	})
	// /slow?d=2s simulates a slow backend, useful for testing timeouts.
	mux.HandleFunc("GET /slow", func(w http.ResponseWriter, r *http.Request) {
		d, err := time.ParseDuration(r.URL.Query().Get("d"))
		if err != nil {
			d = 2 * time.Second
		}
		select {
		case <-time.After(d):
		case <-r.Context().Done():
			return
		}
		fmt.Fprintf(w, "slow response from %s after %s\n", *name, d)
	})
	mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintf(w, "hello from %s (path=%s, x-forwarded-for=%q)\n",
			*name, r.URL.Path, r.Header.Get("X-Forwarded-For"))
	})

	slog.Info("backend listening", "addr", *addr, "name", *name)
	srv := &http.Server{Addr: *addr, Handler: mux, ReadHeaderTimeout: 5 * time.Second}
	if err := srv.ListenAndServe(); err != nil {
		slog.Error("backend stopped", "err", err)
		os.Exit(1)
	}
}

# Go Load Balancer

An HTTP (L7) load balancer built from scratch in Go, as a learning project on the path to infrastructure engineering.

## Layout

```
cmd/lb/            the load balancer binary
cmd/backend/       a tiny demo upstream server (/, /healthz, /slow?d=2s)
internal/config/   JSON config loading + validation
internal/backend/  one upstream server and its health state
internal/balancer/ strategies that choose a backend (round robin, ...)
configs/           example config
.github/workflows/ CI: gofmt, go vet, go test -race, build, golangci-lint
```

## Running locally

`make` isn't required. These are the plain Go commands (they work in Git Bash on Windows):

```bash
# terminal 1-3: start demo backends
go run ./cmd/backend -addr :9001 -name b1
go run ./cmd/backend -addr :9002 -name b2
go run ./cmd/backend -addr :9003 -name b3

# terminal 4: start the load balancer
go run ./cmd/lb -config configs/config.json

# terminal 5: send traffic
for i in 1 2 3 4 5 6; do curl -s localhost:8080; done
```

Checks (the same ones CI runs):

```bash
gofmt -l .
go vet ./...
go test -race ./...    # -race needs cgo/gcc on Windows; CI runs it on Linux
```
